   119→/* ── demo card formatting ───────────────────────────────────────────── */
   120→
   121→const digitsOnly = (v: string) => v.replace(/\D/g, "");
   122→
   123→const formatCardNumber = (v: string) =>
   124→  digitsOnly(v)
   125→    .slice(0, 16)
   126→    .replace(/(\d{4})(?=\d)/g, "$1 ");
   127→
   128→const formatExpiry = (v: string) => {
   129→  const d = digitsOnly(v).slice(0, 4);
   130→  return d.length > 2 ? `${d.slice(0, 2)}/${d.slice(2)}` : d;
   131→};
   132→
   133→function cardError(card: CardForm): string | null {
   134→  const digits = digitsOnly(card.number);
   135→  if (digits.length < 15) return "Ingresa un número de tarjeta válido (demo: 4242…).";
   136→  const [mm, yy] = card.expiry.split("/");
   137→  if (!mm || !yy || Number(mm) < 1 || Number(mm) > 12 || yy.length !== 2)
   138→    return "Vencimiento inválido — usa el formato MM/AA.";
   139→  if (digitsOnly(card.cvv).length < 3) return "CVV inválido.";
   140→  if (card.holder.trim().length < 3) return "Ingresa el nombre del titular.";
   141→  return null;
   142→}
